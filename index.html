#!/usr/bin/env python3 # -*- coding: utf-8 -*- import http.server import socketserver import urllib.request import urllib.parse import ssl PORT = 8080 # لینکە ڕەسەنەکان M3U_URL = "https://xtream-ie.net/get.php?username=serverblbas&password=serverblbas&type=m3u_plus" EPG_URL = "https://xtream-ie.net/xmltv.php?username=serverblbas&password=serverblbas" HTML_PAGE = """ Zaman IPTV
📺 ZAMAN IPTV
0 کەناڵهەموو
    بارکردن...
کەناڵEPG
چاوەڕوان بە...
""" class Handler(http.server.SimpleHTTPRequestHandler): def do_GET(self): if self.path == '/': self.send_response(200) self.send_header('Content-type', 'text/html; charset=utf-8') self.end_headers() # Replace placeholders with actual URLs html = HTML_PAGE.replace('M3U_URL_PLACEHOLDER', M3U_URL).replace('EPG_URL_PLACEHOLDER', EPG_URL) self.wfile.write(html.encode('utf-8')) elif self.path.startswith('/proxy?'): query = urllib.parse.parse_qs(urllib.parse.urlparse(self.path).query) url = query.get('url', [''])[0] if not url: self.send_error(400, "Missing URL") return try: ctx = ssl.create_default_context() ctx.check_hostname = False ctx.verify_mode = ssl.CERT_NONE req = urllib.request.Request(url, headers={ 'User-Agent': 'Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.0', 'Accept': '*/*', 'Accept-Encoding': 'identity', 'Connection': 'keep-alive', }) with urllib.request.urlopen(req, context=ctx, timeout=30) as resp: self.send_response(resp.status) ct = resp.headers.get('Content-Type', 'application/octet-stream') self.send_header('Content-Type', ct) self.send_header('Access-Control-Allow-Origin', '*') self.send_header('Cache-Control', 'no-cache') self.end_headers() self.wfile.write(resp.read()) except Exception as e: self.send_error(500, str(e)) else: self.send_error(404) def log_message(self, format, *args): # Cleaner logs print(f"[{self.log_date_time_string()}] {args[0]}") if __name__ == '__main__': print("=" * 50) print(" ZAMAN IPTV - Local Server") print("=" * 50) print(f"\n👉 Open: http://localhost:{PORT}") print("👉 Press Ctrl+C to stop\n") socketserver.TCPServer.allow_reuse_address = True with socketserver.TCPServer(("", PORT), Handler) as httpd: try: httpd.serve_forever() except KeyboardInterrupt: print("\n\nServer stopped.")
